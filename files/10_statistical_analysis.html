1 Purpose of this stage

Stage 9 produced point estimates of value-weighted CRC risk and intervention rate for both selected models (XGBoost, Elastic-Net), at every alpha, on iid_holdout and future_test. Point estimates alone don’t tell you whether one model’s number being lower is a real effect or sampling noise. This stage quantifies that, using the same paired-bootstrap approach Stage 8 already established for the AUC comparison: resample rows once per replicate, score both models on the identical resample, so the comparison controls for shared sampling variance rather than treating each model’s uncertainty as independent.

What this adds, concretely:

  • A 95% bootstrap CI on the risk difference (XGBoost minus Elastic-Net) at every alpha, on both iid_holdout and future_test.
  • A 95% bootstrap CI on the intervention-rate difference, same structure, on both splits.
  • Closed-form Wilson CIs on each model’s intervention rate itself (a proportion, so Wilson is the more defensible interval than a normal approximation at these sample sizes) – reusing wilson_ci() from 00_setup.R directly, not reimplementing it.
  • An explicit significance verdict per alpha per split: whether 0 falls inside the bootstrapped CI of the difference.

Thresholds (tau) are fixed at the values Stage 9 already selected on crc_calibration – this stage resamples the evaluation rows only, not the threshold-selection process. That is deliberate: it answers “given this policy, how uncertain is our estimate of its risk/workload on fresh data,” which is the question a final report needs, not “how would the threshold itself vary under resampling” (a different, noisier question Stage 9’s CRC guarantee already handles formally for the calibration side).

1.1 Scoring must match Stage 9 exactly

The calibrators saved by Stage 9 were fit on the raw probabilities produced by Stage 9’s scorers. This stage therefore uses the same scorers, copied verbatim: the XGBoost booster is loaded from xgboost_tuned.ubj (never from the .rds Booster, whose external pointer is stale across sessions), its trees are read out and scored in pure R, and its design matrix is built from the booster’s own feature names. See Stage 9 for why xgboost’s native predict cannot be used on this machine.

load_split_from_db <- function(con, split_name, preprocessor, feature_selector, bin_levels) {
  dt <- as.data.table(DBI::dbGetQuery(con, paste0("SELECT * FROM ", split_name)))
  for (col in feature_selector$categorical_features) {
    if (col %in% names(dt)) {
      levs <- union(preprocessor$allowed_levels[[col]], "OTHER")
      set(dt, j = col, value = factor(dt[[col]], levels = levs))
    }
  }
  for (col in names(bin_levels)) {
    if (col %in% names(dt)) set(dt, j = col, value = factor(dt[[col]], levels = bin_levels[[col]]))
  }
  dt
}

build_onehot <- function(dt, predictors) {
  model.matrix(as.formula(paste("~", paste(predictors, collapse = " + "), "- 1")), data = dt)
}

# Pads any training columns missing from x with zeros and reorders to the
# training column order. Always returns a plain matrix (never a data.frame).
align_onehot <- function(x, train_colnames) {
  x <- as.matrix(x)
  missing_cols <- setdiff(train_colnames, colnames(x))
  if (length(missing_cols) > 0) {
    add <- matrix(0, nrow = nrow(x), ncol = length(missing_cols),
                  dimnames = list(NULL, missing_cols))
    x <- cbind(x, add)
  }
  x[, train_colnames, drop = FALSE]
}

# ---- Pure-R scoring of an xgboost booster -------------------------------------
xgb_extract_trees <- function(booster, wrapper_colnames = NULL) {
  dt <- as.data.table(xgboost::xgb.model.dt.tree(model = booster))
  leaf_col <- if ("Gain" %in% names(dt)) "Gain" else "Quality"
  feat_chr <- as.character(dt$Feature)
  split_feats <- unique(feat_chr[feat_chr != "Leaf"])

  unnamed <- all(grepl("^f[0-9]+$", split_feats)) &&
    !is.null(wrapper_colnames) && !all(split_feats %in% wrapper_colnames)
  if (unnamed) {
    feat_names <- wrapper_colnames
    fidx <- as.integer(sub("^f", "", feat_chr)) + 1L
  } else {
    feat_names <- split_feats
    fidx <- match(feat_chr, feat_names)
  }
  fidx[feat_chr == "Leaf"] <- 0L

  trees <- lapply(split(seq_len(nrow(dt)), dt$Tree), function(rows) {
    d <- dt[rows]; fi <- fidx[rows]
    o <- order(d$Node); d <- d[o]; fi <- fi[o]
    stopifnot(identical(as.integer(d$Node), seq_len(nrow(d)) - 1L))
    list(feat  = as.integer(fi),
         split = as.numeric(d$Split),
         yes   = match(d$Yes, d$ID) - 1L,
         no    = match(d$No,  d$ID) - 1L,
         value = as.numeric(d[[leaf_col]]))
  })
  list(trees = trees, feat_names = feat_names)
}

xgb_base_margin <- function(booster) {
  bs <- tryCatch({
    cfg <- xgboost::xgb.config(booster)
    v <- NULL
    if (is.list(cfg)) v <- cfg$learner$learner_model_param$base_score
    if (is.null(v) && is.character(cfg))
      v <- sub('.*"base_score"\\s*:\\s*"([^"]+)".*', "\\1", cfg)
    as.numeric(gsub("[][]", "", v))[1]
  }, error = function(e) NA_real_)
  if (is.na(bs)) {
    warning("Could not read base_score from the booster config; assuming 0.5.")
    bs <- 0.5
  }
  qlogis(bs)
}

# Splits are "go to Yes branch if value < Split", as in xgboost.
xgb_predict_trees <- function(fit, x) {
  if (anyNA(x)) stop("NA in model matrix; tree scorer does not handle missing values.")
  n <- nrow(x)
  margin <- rep(fit$base_margin, n)
  for (tr in fit$trees) {
    pos <- rep.int(0L, n)
    repeat {
      f <- tr$feat[pos + 1L]
      act <- which(f > 0L)
      if (length(act) == 0L) break
      xv <- x[cbind(act, f[act])]
      p <- pos[act] + 1L
      pos[act] <- ifelse(xv < tr$split[p], tr$yes[p], tr$no[p])
    }
    margin <- margin + tr$value[pos + 1L]
  }
  plogis(margin)
}

# ---- Design matrix built from the booster's OWN feature names ------------------
resolve_design_spec <- function(dt, feat_names) {
  vars <- names(dt)
  spec <- lapply(feat_names, function(f) {
    if (f %in% vars && (is.numeric(dt[[f]]) || is.logical(dt[[f]])))
      return(list(name = f, var = f, level = NULL))
    cand <- vars[startsWith(f, vars) & nchar(f) > nchar(vars)]
    if (length(cand) == 0) return(list(name = f, var = NA_character_, level = NULL))
    v <- cand[which.max(nchar(cand))]
    list(name = f, var = v, level = substring(f, nchar(v) + 1L))
  })
  unresolved <- feat_names[vapply(spec, function(z) is.na(z$var), logical(1))]
  if (length(unresolved) > 0) {
    stop(length(unresolved), " booster feature(s) cannot be built from the data columns: ",
         paste(head(unresolved, 8), collapse = ", "))
  }
  spec
}

build_design <- function(dt, spec) {
  x <- matrix(0, nrow = nrow(dt), ncol = length(spec),
              dimnames = list(NULL, vapply(spec, function(z) z$name, character(1))))
  for (j in seq_along(spec)) {
    z <- spec[[j]]
    col <- dt[[z$var]]
    x[, j] <- if (is.null(z$level)) as.numeric(col) else as.numeric(as.character(col) %in% z$level)
  }
  x
}

load_xgb_robust <- function(entry) {
  # Metadata from the RDS wrapper; its Booster is NOT used (stale external pointer).
  wrapper <- readRDS(entry$model_path)
  predictors <- wrapper$predictors
  wrapper_colnames <- wrapper$train_colnames
  rm(wrapper)

  booster <- xgboost::xgb.load(entry$xgb_ubj_path)
  ex <- xgb_extract_trees(booster, wrapper_colnames)
  base_margin <- xgb_base_margin(booster)
  cat("xgboost booster loaded from .ubj: ", length(ex$trees), " trees, ",
      length(ex$feat_names), " split features, base_score = ",
      round(plogis(base_margin), 5), ".\n", sep = "")

  list(model = booster, trees = ex$trees, base_margin = base_margin,
       predictors = predictors, train_colnames = ex$feat_names)
}

predict_xgboost <- function(fit, newdata, chunk_size = PRED_CHUNK_SIZE) {
  n <- nrow(newdata)
  spec <- resolve_design_spec(newdata, fit$train_colnames)
  out <- numeric(n)
  for (s in seq(1, n, by = chunk_size)) {
    e <- min(s + chunk_size - 1, n)
    out[s:e] <- xgb_predict_trees(fit, build_design(newdata[s:e], spec))
  }
  out
}

predict_glmnet_selected <- function(fit, newdata, chunk_size = PRED_CHUNK_SIZE) {
  n <- nrow(newdata)
  out <- numeric(n)
  for (s in seq(1, n, by = chunk_size)) {
    e <- min(s + chunk_size - 1, n)
    x <- align_onehot(build_onehot(newdata[s:e], fit$predictors), fit$train_colnames)
    out[s:e] <- as.numeric(predict(fit$model, newx = x, s = fit$lambda, type = "response"))
  }
  out
}

# Paired bootstrap on PRE-COMPUTED per-row vectors: one set of resampled row
# indices per replicate is applied to both models' vectors, so the comparison
# controls for shared sampling variance.
paired_bootstrap_diff <- function(a_vec, b_vec, n_boot = N_BOOT) {
  n <- length(a_vec)
  diffs <- numeric(n_boot); vals_a <- numeric(n_boot); vals_b <- numeric(n_boot)
  for (b in seq_len(n_boot)) {
    idx <- sample.int(n, n, replace = TRUE)
    va <- mean(a_vec[idx]); vb <- mean(b_vec[idx])
    vals_a[b] <- va; vals_b[b] <- vb; diffs[b] <- va - vb
  }
  ci <- quantile(diffs, c(0.025, 0.975), names = FALSE)
  data.table(mean_a = mean(vals_a), mean_b = mean(vals_b), mean_diff = mean(diffs),
             diff_ci_low = ci[1], diff_ci_high = ci[2],
             significant = (ci[1] > 0) || (ci[2] < 0))
}

# Risk and intervention-rate comparison, XGBoost (a) vs. Elastic-Net (b),
# at every alpha, with tau fixed at the Stage 9 values.
compare_models <- function(y, v, probs, split_name) {
  rbindlist(lapply(ALPHA_GRID, function(a) {
    tau_x <- global_crc_results[model_id == "xgboost" & alpha == a, tau]
    tau_e <- global_crc_results[model_id == "elastic_net_logistic" & alpha == a, tau]

    risk <- paired_bootstrap_diff(y * v * (probs$xgboost < tau_x),
                                  y * v * (probs$elastic_net_logistic < tau_e))
    risk[, `:=`(alpha = a, split = split_name, metric = "risk")]

    interv <- paired_bootstrap_diff(as.numeric(probs$xgboost >= tau_x),
                                    as.numeric(probs$elastic_net_logistic >= tau_e))
    interv[, `:=`(alpha = a, split = split_name, metric = "intervention_rate")]

    rbindlist(list(risk, interv))
  }))[, `:=`(xgboost_value = mean_a, elastic_net_value = mean_b)]
}

# Purpose of this stage

Stage 9 produced point estimates of value-weighted CRC risk and intervention rate for both selected models (XGBoost, Elastic-Net), at every alpha, on `iid_holdout` and `future_test`. Point estimates alone don't tell you whether one model's number being lower is a real effect or sampling noise. This stage quantifies that, using the same paired-bootstrap approach Stage 8 already established for the AUC comparison: resample rows once per replicate, score both models on the identical resample, so the comparison controls for shared sampling variance rather than treating each model's uncertainty as independent.

What this adds, concretely:

- A 95% bootstrap CI on the **risk difference** (XGBoost minus Elastic-Net) at every alpha, on both `iid_holdout` and `future_test`.
- A 95% bootstrap CI on the **intervention-rate difference**, same structure, on both splits.
- Closed-form **Wilson CIs** on each model's intervention rate itself (a proportion, so Wilson is the more defensible interval than a normal approximation at these sample sizes) -- reusing `wilson_ci()` from `00_setup.R` directly, not reimplementing it.
- An explicit **significance verdict** per alpha per split: whether 0 falls inside the bootstrapped CI of the difference.

Thresholds (tau) are fixed at the values Stage 9 already selected on `crc_calibration` -- this stage resamples the evaluation rows only, not the threshold-selection process. That is deliberate: it answers "given this policy, how uncertain is our estimate of its risk/workload on fresh data," which is the question a final report needs, not "how would the threshold itself vary under resampling" (a different, noisier question Stage 9's CRC guarantee already handles formally for the calibration side).

## Scoring must match Stage 9 exactly

The calibrators saved by Stage 9 were fit on the raw probabilities produced by Stage 9's scorers. This stage therefore uses the **same** scorers, copied verbatim: the XGBoost booster is loaded from `xgboost_tuned.ubj` (never from the `.rds` Booster, whose external pointer is stale across sessions), its trees are read out and scored in pure R, and its design matrix is built from the booster's own feature names. See Stage 9 for why xgboost's native predict cannot be used on this machine.

```{r scorers}
build_onehot <- function(dt, predictors) {
  model.matrix(as.formula(paste("~", paste(predictors, collapse = " + "), "- 1")), data = dt)
}

# Pads any training columns missing from x with zeros and reorders to the
# training column order. Always returns a plain matrix (never a data.frame).
align_onehot <- function(x, train_colnames) {
  x <- as.matrix(x)
  missing_cols <- setdiff(train_colnames, colnames(x))
  if (length(missing_cols) > 0) {
    add <- matrix(0, nrow = nrow(x), ncol = length(missing_cols),
                  dimnames = list(NULL, missing_cols))
    x <- cbind(x, add)
  }
  x[, train_colnames, drop = FALSE]
}

# ---- Pure-R scoring of an xgboost booster -------------------------------------
xgb_extract_trees <- function(booster, wrapper_colnames = NULL) {
  dt <- as.data.table(xgboost::xgb.model.dt.tree(model = booster))
  leaf_col <- if ("Gain" %in% names(dt)) "Gain" else "Quality"
  feat_chr <- as.character(dt$Feature)
  split_feats <- unique(feat_chr[feat_chr != "Leaf"])

  unnamed <- all(grepl("^f[0-9]+$", split_feats)) &&
    !is.null(wrapper_colnames) && !all(split_feats %in% wrapper_colnames)
  if (unnamed) {
    feat_names <- wrapper_colnames
    fidx <- as.integer(sub("^f", "", feat_chr)) + 1L
  } else {
    feat_names <- split_feats
    fidx <- match(feat_chr, feat_names)
  }
  fidx[feat_chr == "Leaf"] <- 0L

  trees <- lapply(split(seq_len(nrow(dt)), dt$Tree), function(rows) {
    d <- dt[rows]; fi <- fidx[rows]
    o <- order(d$Node); d <- d[o]; fi <- fi[o]
    stopifnot(identical(as.integer(d$Node), seq_len(nrow(d)) - 1L))
    list(feat  = as.integer(fi),
         split = as.numeric(d$Split),
         yes   = match(d$Yes, d$ID) - 1L,
         no    = match(d$No,  d$ID) - 1L,
         value = as.numeric(d[[leaf_col]]))
  })
  list(trees = trees, feat_names = feat_names)
}

xgb_base_margin <- function(booster) {
  bs <- tryCatch({
    cfg <- xgboost::xgb.config(booster)
    v <- NULL
    if (is.list(cfg)) v <- cfg$learner$learner_model_param$base_score
    if (is.null(v) && is.character(cfg))
      v <- sub('.*"base_score"\\s*:\\s*"([^"]+)".*', "\\1", cfg)
    as.numeric(gsub("[][]", "", v))[1]
  }, error = function(e) NA_real_)
  if (is.na(bs)) {
    warning("Could not read base_score from the booster config; assuming 0.5.")
    bs <- 0.5
  }
  qlogis(bs)
}

# Splits are "go to Yes branch if value < Split", as in xgboost.
xgb_predict_trees <- function(fit, x) {
  if (anyNA(x)) stop("NA in model matrix; tree scorer does not handle missing values.")
  n <- nrow(x)
  margin <- rep(fit$base_margin, n)
  for (tr in fit$trees) {
    pos <- rep.int(0L, n)
    repeat {
      f <- tr$feat[pos + 1L]
      act <- which(f > 0L)
      if (length(act) == 0L) break
      xv <- x[cbind(act, f[act])]
      p <- pos[act] + 1L
      pos[act] <- ifelse(xv < tr$split[p], tr$yes[p], tr$no[p])
    }
    margin <- margin + tr$value[pos + 1L]
  }
  plogis(margin)
}

# ---- Design matrix built from the booster's OWN feature names ------------------
resolve_design_spec <- function(dt, feat_names) {
  vars <- names(dt)
  spec <- lapply(feat_names, function(f) {
    if (f %in% vars && (is.numeric(dt[[f]]) || is.logical(dt[[f]])))
      return(list(name = f, var = f, level = NULL))
    cand <- vars[startsWith(f, vars) & nchar(f) > nchar(vars)]
    if (length(cand) == 0) return(list(name = f, var = NA_character_, level = NULL))
    v <- cand[which.max(nchar(cand))]
    list(name = f, var = v, level = substring(f, nchar(v) + 1L))
  })
  unresolved <- feat_names[vapply(spec, function(z) is.na(z$var), logical(1))]
  if (length(unresolved) > 0) {
    stop(length(unresolved), " booster feature(s) cannot be built from the data columns: ",
         paste(head(unresolved, 8), collapse = ", "))
  }
  spec
}

build_design <- function(dt, spec) {
  x <- matrix(0, nrow = nrow(dt), ncol = length(spec),
              dimnames = list(NULL, vapply(spec, function(z) z$name, character(1))))
  for (j in seq_along(spec)) {
    z <- spec[[j]]
    col <- dt[[z$var]]
    x[, j] <- if (is.null(z$level)) as.numeric(col) else as.numeric(as.character(col) %in% z$level)
  }
  x
}

load_xgb_robust <- function(entry) {
  # Metadata from the RDS wrapper; its Booster is NOT used (stale external pointer).
  wrapper <- readRDS(entry$model_path)
  predictors <- wrapper$predictors
  wrapper_colnames <- wrapper$train_colnames
  rm(wrapper)

  booster <- xgboost::xgb.load(entry$xgb_ubj_path)
  ex <- xgb_extract_trees(booster, wrapper_colnames)
  base_margin <- xgb_base_margin(booster)
  cat("xgboost booster loaded from .ubj: ", length(ex$trees), " trees, ",
      length(ex$feat_names), " split features, base_score = ",
      round(plogis(base_margin), 5), ".\n", sep = "")

  list(model = booster, trees = ex$trees, base_margin = base_margin,
       predictors = predictors, train_colnames = ex$feat_names)
}

predict_xgboost <- function(fit, newdata, chunk_size = PRED_CHUNK_SIZE) {
  n <- nrow(newdata)
  spec <- resolve_design_spec(newdata, fit$train_colnames)
  out <- numeric(n)
  for (s in seq(1, n, by = chunk_size)) {
    e <- min(s + chunk_size - 1, n)
    out[s:e] <- xgb_predict_trees(fit, build_design(newdata[s:e], spec))
  }
  out
}

predict_glmnet_selected <- function(fit, newdata, chunk_size = PRED_CHUNK_SIZE) {
```

```{r helpers}
# Paired bootstrap on PRE-COMPUTED per-row vectors: one set of resampled row
# indices per replicate is applied to both models' vectors, so the comparison
# controls for shared sampling variance.
paired_bootstrap_diff <- function(a_vec, b_vec, n_boot = N_BOOT) {
  n <- length(a_vec)
  diffs <- numeric(n_boot); vals_a <- numeric(n_boot); vals_b <- numeric(n_boot)
  for (b in seq_len(n_boot)) {
    idx <- sample.int(n, n, replace = TRUE)
    va <- mean(a_vec[idx]); vb <- mean(b_vec[idx])
    vals_a[b] <- va; vals_b[b] <- vb; diffs[b] <- va - vb
  }
  ci <- quantile(diffs, c(0.025, 0.975), names = FALSE)
  data.table(mean_a = mean(vals_a), mean_b = mean(vals_b), mean_diff = mean(diffs),
             diff_ci_low = ci[1], diff_ci_high = ci[2],
             significant = (ci[1] > 0) || (ci[2] < 0))
}

# Risk and intervention-rate comparison, XGBoost (a) vs. Elastic-Net (b),
# at every alpha, with tau fixed at the Stage 9 values.
compare_models <- function(y, v, probs, split_name) {
  rbindlist(lapply(ALPHA_GRID, function(a) {
    tau_x <- global_crc_results[model_id == "xgboost" & alpha == a, tau]
    tau_e <- global_crc_results[model_id == "elastic_net_logistic" & alpha == a, tau]

    risk <- paired_bootstrap_diff(y * v * (probs$xgboost < tau_x),
                                  y * v * (probs$elastic_net_logistic < tau_e))
    risk[, `:=`(alpha = a, split = split_name, metric = "risk")]

    interv <- paired_bootstrap_diff(as.numeric(probs$xgboost >= tau_x),
                                    as.numeric(probs$elastic_net_logistic >= tau_e))
    interv[, `:=`(alpha = a, split = split_name, metric = "intervention_rate")]

    rbindlist(list(risk, interv))
  }))[, `:=`(xgboost_value = mean_a, elastic_net_value = mean_b)]
}
```

```{r wilson-ci}
    tau <- global_crc_results[model_id == id & alpha == a, tau]
    n_intervened <- sum(holdout_probs[[id]] >= tau)
    n <- length(holdout_probs[[id]])
    wci <- wilson_ci(n_intervened, n)
    data.table(model_id = id, alpha = a, tau = tau, n_intervened = n_intervened, n = n,
               rate = wci$p_hat, ci_low = wci$lower, ci_high = wci$upper)
  }))
}))
fwrite(wilson_results, file.path(TABLE_DIR, "s10_01_wilson_ci_intervention_rate.csv"))
wilson_results
```
