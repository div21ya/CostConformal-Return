1 Purpose of this stage

Stage 6 fits 14 algorithms unconditionally, plus LightGBM as a 15th if it is installed (see “Package availability” below) – satisfying Phase 2’s 10-15-algorithm requirement comfortably either way. Data comes from Stage 5’s SQLite database via load_split_from_db(), not from raw *_features.csv.gz reads, and predictor sets come from Stage 4’s r_feature_selector.rds (linear_features for linear/distance models, tree_features for tree/boosting models).

Each model is fit on model_train, scored on validation, and both the fitted model object and its validation predictions are saved – this stage’s job is implementation, not tuning (Stage 7) or final comparison (Stage 8), so default/reasonable hyperparameters are used throughout.

1.1 Memory and robustness profile (6 GB RAM)

  • model_train is frozen to a 250k-row pool once, immediately after loading, and the full 890k-row table is freed. No model in this stage trains on more than DEFAULT_TRAIN_CAP (250,000) rows, and every 250k-capped model already called set.seed(20260917) right before sampling from the same source table – so all nine of them were already receiving an identical sample. Holding 890k rows resident for the whole stage bought nothing; this is a pure memory fix with zero effect on those nine models, and only a minor (still properly isReturned-stratified) change of sampling pool for the smaller-cap models below.
  • KNN’s caps are deliberately much smaller than the others (8,000 train / 5,000 scored). kknn computes a full train x test distance structure – at the previous 30k/20k caps that’s a 30,000 x 20,000 array of doubles, roughly 4.8 GB on its own. This was the single biggest risk in the original design. At 8,000 x 5,000 that drops to ~320 MB.
  • SVM’s caps (15,000 / 15,000) are intentionally left larger than KNN’s. e1071::svm (libsvm) bounds its kernel cache explicitly rather than materializing an O(n_train x n_test) array the way kknn does, so it was never the same category of risk – shrinking it further would only cost accuracy for no real memory benefit.
  • Random Forest / Extra Trees now use sample.fraction = 0.5 (each of num.trees trees bootstraps half the data, not all of it) and num.trees is reduced from 300 to 200 – standard subbagging, trims per-tree memory without materially hurting the ensemble.
  • XGBoost / LightGBM cap max_bin at 128 (default 256) – histogram memory scales with this directly.
  • GBM / Neural Net caps trimmed further (150k->100k, 100k->60k) as cheap extra headroom, since both had less to lose in exchange for the smaller sample than KNN/SVM did.
  • One model fit at a time, training subsample/design matrix/fitted object freed (rm() + gc()) before the next model starts; every model fit is wrapped in tryCatch so one failure doesn’t take down the rest.

1.2 Package availability

glmnet, MASS, naivebayes, kknn, rpart, ranger, gbm, xgboost, e1071, nnet are required (MASS/rpart/nnet ship with base R’s recommended-package set and are usually already present). lightgbm is optional – if requireNamespace("lightgbm", quietly = TRUE) fails, that one model is skipped with a clear note, rather than failing the whole stage.

load_split_from_db <- function(con, split_name, preprocessor, feature_selector, bin_levels) {
  dt <- as.data.table(DBI::dbGetQuery(con, paste0("SELECT * FROM ", split_name)))
  for (col in feature_selector$categorical_features) {
    if (col %in% names(dt)) {
      levs <- union(preprocessor$allowed_levels[[col]], "OTHER")
      set(dt, j = col, value = factor(dt[[col]], levels = levs))
    }
  }
  for (col in names(bin_levels)) {
    if (col %in% names(dt)) set(dt, j = col, value = factor(dt[[col]], levels = bin_levels[[col]]))
  }
  dt
}

stratified_sample <- function(dt, max_n) {
  if (nrow(dt) <= max_n) return(dt)
  frac <- max_n / nrow(dt)
  dt[, .SD[sample(.N, size = max(1, round(.N * frac)))], by = isReturned]
}

build_onehot <- function(dt, predictors) {
  model.matrix(as.formula(paste("~", paste(predictors, collapse = " + "), "- 1")), data = dt)
}

align_onehot <- function(x, train_colnames) {
  missing_cols <- setdiff(train_colnames, colnames(x))
  for (mc in missing_cols) x <- cbind(x, stats::setNames(data.frame(rep(0, nrow(x))), mc))
  x[, train_colnames, drop = FALSE]
}

fit_numeric_scale <- function(dt, numeric_cols) {
  list(
    center = sapply(numeric_cols, function(c) mean(dt[[c]], na.rm = TRUE)),
    scale  = sapply(numeric_cols, function(c) { s <- sd(dt[[c]], na.rm = TRUE); if (s == 0) 1 else s })
  )
}
apply_numeric_scale <- function(dt, sp) {
  dt <- copy(dt)
  for (c in names(sp$center)) set(dt, j = c, value = (dt[[c]] - sp$center[[c]]) / sp$scale[[c]])
  dt
}

compute_metrics <- function(y, p) {
  p <- pmin(pmax(p, 1e-6), 1 - 1e-6)
  pred_class <- as.integer(p >= 0.5)
  tp <- sum(pred_class == 1 & y == 1); fp <- sum(pred_class == 1 & y == 0)
  tn <- sum(pred_class == 0 & y == 0); fn <- sum(pred_class == 0 & y == 1)
  precision <- if ((tp + fp) == 0) NA_real_ else tp / (tp + fp)
  recall <- if ((tp + fn) == 0) NA_real_ else tp / (tp + fn)
  f1 <- if (is.na(precision) || is.na(recall) || (precision + recall) == 0) NA_real_ else 2 * precision * recall / (precision + recall)
  data.table(
    auc = auc_roc(y, p), pr_auc = pr_auc(y, p), brier = brier_score(y, p), log_loss = log_loss(y, p),
    accuracy = (tp + tn) / length(y), precision = precision, recall = recall, f1 = f1
  )
}

run_model <- function(id, family, fit_fn, predict_fn, train_cap, val_cap = NULL) {
  cat("\n=== Fitting:", id, "===\n")
  set.seed(20260917)
  train_sub <- stratified_sample(model_train, train_cap)

  val_idx <- if (!is.null(val_cap) && nrow(validation) > val_cap) {
    sort(sample.int(nrow(validation), val_cap))
  } else {
    seq_len(nrow(validation))
  }
  val_sub <- validation[val_idx]

  t0 <- Sys.time()
  fit <- fit_fn(train_sub)
  fit_seconds <- as.numeric(difftime(Sys.time(), t0, units = "secs"))

  prob <- predict_fn(fit, val_sub)
  metrics <- compute_metrics(val_sub$isReturned, prob)
  metrics[, `:=`(model_id = id, family = family, train_n = nrow(train_sub), val_n = nrow(val_sub), fit_seconds = fit_seconds)]

  saveRDS(fit, file.path(MODEL_ZOO_DIR, paste0(id, ".rds")))
  cat("Done:", id, "-- AUC:", round(metrics$auc, 4), "| fit time:", round(fit_seconds, 1), "s\n")

  rm(train_sub, val_sub, fit)
  invisible(gc(verbose = FALSE))
  list(metrics = metrics, prob = prob, idx = val_idx)
}

fit_logistic <- function(train_sub) {
  glm(as.formula(paste("isReturned ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
      data = train_sub, family = binomial())
}
predict_logistic <- function(fit, newdata) as.numeric(predict(fit, newdata = newdata, type = "response"))

fit_glmnet_selected <- function(train_sub, alpha) {
  n <- nrow(train_sub)
  tune_idx <- sample.int(n, size = round(0.2 * n))
  fit_idx <- setdiff(seq_len(n), tune_idx)

  x_fit <- build_onehot(train_sub[fit_idx], LINEAR_PREDICTORS)
  y_fit <- train_sub$isReturned[fit_idx]
  path <- glmnet(x_fit, y_fit, family = "binomial", alpha = alpha, standardize = TRUE)

  x_tune <- build_onehot(train_sub[tune_idx], LINEAR_PREDICTORS)
  y_tune <- train_sub$isReturned[tune_idx]
  pred_path <- predict(path, newx = x_tune, type = "response")
  briers <- apply(pred_path, 2, function(p) brier_score(y_tune, p))
  best_lambda <- path$lambda[which.min(briers)]

  x_full <- build_onehot(train_sub, LINEAR_PREDICTORS)
  final_model <- glmnet(x_full, train_sub$isReturned, family = "binomial", alpha = alpha, lambda = best_lambda, standardize = TRUE)
  list(model = final_model, lambda = best_lambda, predictors = LINEAR_PREDICTORS, train_colnames = colnames(x_full))
}
predict_glmnet_selected <- function(fit, newdata) {
  x <- align_onehot(build_onehot(newdata, fit$predictors), fit$train_colnames)
  as.numeric(predict(fit$model, newx = x, s = fit$lambda, type = "response"))
}

fit_lda <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  MASS::lda(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))), data = train_sub)
}
predict_lda <- function(fit, newdata) as.numeric(predict(fit, newdata = newdata)$posterior[, "Yes"])

fit_naive_bayes <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  naivebayes::naive_bayes(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                           data = train_sub, usekernel = TRUE)
}
predict_naive_bayes <- function(fit, newdata) as.numeric(predict(fit, newdata = newdata, type = "prob")[, "Yes"])

# KNN is a lazy learner -- the "fit" is the frozen training subsample + k;
# kknn() both fits and predicts in one call. See the memory-profile note
# above for why its caps are much smaller than every other model's.
fit_knn <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  list(train_data = train_sub, k = 15)
}
predict_knn <- function(fit, newdata) {
  nd <- copy(newdata)
  nd[, y_factor := factor(NA, levels = c("No", "Yes"))]
  res <- kknn::kknn(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                     train = fit$train_data, test = nd, k = fit$k, kernel = "optimal", scale = TRUE)
  as.numeric(res$prob[, "Yes"])
}

fit_svm <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  e1071::svm(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
             data = train_sub, kernel = "linear", probability = TRUE, scale = TRUE)
}
predict_svm <- function(fit, newdata) {
  pred <- predict(fit, newdata = newdata, probability = TRUE)
  as.numeric(attr(pred, "probabilities")[, "Yes"])
}

fit_nnet <- function(train_sub) {
  scale_params <- fit_numeric_scale(train_sub, feature_selector$linear_numeric_features)
  train_scaled <- apply_numeric_scale(train_sub, scale_params)
  model <- nnet::nnet(as.formula(paste("isReturned ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                       data = train_scaled, size = 8, decay = 0.01, maxit = 200, linout = FALSE, trace = FALSE)
  list(model = model, scale_params = scale_params)
}
predict_nnet <- function(fit, newdata) {
  nd_scaled <- apply_numeric_scale(newdata, fit$scale_params)
  as.numeric(predict(fit$model, newdata = nd_scaled, type = "raw"))
}

}

    run_model(spec$id, spec$family, spec$fit_fn, spec$predict_fn, spec$train_cap, spec$val_cap),

# Purpose of this stage

Stage 6 fits 14 algorithms unconditionally, plus LightGBM as a 15th if it is installed (see "Package availability" below) -- satisfying Phase 2's 10-15-algorithm requirement comfortably either way. Data comes from Stage 5's SQLite database via `load_split_from_db()`, not from raw `*_features.csv.gz` reads, and predictor sets come from Stage 4's `r_feature_selector.rds` (`linear_features` for linear/distance models, `tree_features` for tree/boosting models).

Each model is fit on `model_train`, scored on `validation`, and both the fitted model object and its validation predictions are saved -- this stage's job is *implementation*, not tuning (Stage 7) or final comparison (Stage 8), so default/reasonable hyperparameters are used throughout.

## Memory and robustness profile (6 GB RAM)

- **`model_train` is frozen to a 250k-row pool once, immediately after loading, and the full 890k-row table is freed.** No model in this stage trains on more than `DEFAULT_TRAIN_CAP` (250,000) rows, and every 250k-capped model already called `set.seed(20260917)` right before sampling from the *same* source table -- so all nine of them were already receiving an identical sample. Holding 890k rows resident for the whole stage bought nothing; this is a pure memory fix with zero effect on those nine models, and only a minor (still properly `isReturned`-stratified) change of sampling *pool* for the smaller-cap models below.
- **KNN's caps are deliberately much smaller than the others (8,000 train / 5,000 scored).** `kknn` computes a full train x test distance structure -- at the previous 30k/20k caps that's a 30,000 x 20,000 array of doubles, roughly **4.8 GB on its own**. This was the single biggest risk in the original design. At 8,000 x 5,000 that drops to \~320 MB.
- **SVM's caps (15,000 / 15,000) are intentionally left larger than KNN's.** `e1071::svm` (libsvm) bounds its kernel cache explicitly rather than materializing an O(n_train x n_test) array the way `kknn` does, so it was never the same category of risk -- shrinking it further would only cost accuracy for no real memory benefit.
- **Random Forest / Extra Trees** now use `sample.fraction = 0.5` (each of `num.trees` trees bootstraps half the data, not all of it) and `num.trees` is reduced from 300 to 200 -- standard subbagging, trims per-tree memory without materially hurting the ensemble.
- **XGBoost / LightGBM** cap `max_bin` at 128 (default 256) -- histogram memory scales with this directly.
- **GBM / Neural Net** caps trimmed further (150k-\>100k, 100k-\>60k) as cheap extra headroom, since both had less to lose in exchange for the smaller sample than KNN/SVM did.
- **One model fit at a time**, training subsample/design matrix/fitted object freed (`rm()` + `gc()`) before the next model starts; every model fit is wrapped in `tryCatch` so one failure doesn't take down the rest.

## Package availability

`glmnet`, `MASS`, `naivebayes`, `kknn`, `rpart`, `ranger`, `gbm`, `xgboost`, `e1071`, `nnet` are required (`MASS`/`rpart`/`nnet` ship with base R's recommended-package set and are usually already present). **`lightgbm` is optional** -- if `requireNamespace("lightgbm", quietly = TRUE)` fails, that one model is skipped with a clear note, rather than failing the whole stage.

```{r shared-helpers}
stratified_sample <- function(dt, max_n) {
  if (nrow(dt) <= max_n) return(dt)
  frac <- max_n / nrow(dt)
  dt[, .SD[sample(.N, size = max(1, round(.N * frac)))], by = isReturned]
}

build_onehot <- function(dt, predictors) {
  model.matrix(as.formula(paste("~", paste(predictors, collapse = " + "), "- 1")), data = dt)
}

align_onehot <- function(x, train_colnames) {
  missing_cols <- setdiff(train_colnames, colnames(x))
  for (mc in missing_cols) x <- cbind(x, stats::setNames(data.frame(rep(0, nrow(x))), mc))
  x[, train_colnames, drop = FALSE]
}

fit_numeric_scale <- function(dt, numeric_cols) {
  list(
    center = sapply(numeric_cols, function(c) mean(dt[[c]], na.rm = TRUE)),
    scale  = sapply(numeric_cols, function(c) { s <- sd(dt[[c]], na.rm = TRUE); if (s == 0) 1 else s })
  )
}
apply_numeric_scale <- function(dt, sp) {
  dt <- copy(dt)
  for (c in names(sp$center)) set(dt, j = c, value = (dt[[c]] - sp$center[[c]]) / sp$scale[[c]])
  dt
}

compute_metrics <- function(y, p) {
  p <- pmin(pmax(p, 1e-6), 1 - 1e-6)
  pred_class <- as.integer(p >= 0.5)
  tp <- sum(pred_class == 1 & y == 1); fp <- sum(pred_class == 1 & y == 0)
  tn <- sum(pred_class == 0 & y == 0); fn <- sum(pred_class == 0 & y == 1)
  precision <- if ((tp + fp) == 0) NA_real_ else tp / (tp + fp)
  recall <- if ((tp + fn) == 0) NA_real_ else tp / (tp + fn)
  f1 <- if (is.na(precision) || is.na(recall) || (precision + recall) == 0) NA_real_ else 2 * precision * recall / (precision + recall)
  data.table(
    auc = auc_roc(y, p), pr_auc = pr_auc(y, p), brier = brier_score(y, p), log_loss = log_loss(y, p),
    accuracy = (tp + tn) / length(y), precision = precision, recall = recall, f1 = f1
  )
}

run_model <- function(id, family, fit_fn, predict_fn, train_cap, val_cap = NULL) {
  cat("\n=== Fitting:", id, "===\n")
  set.seed(20260917)
  train_sub <- stratified_sample(model_train, train_cap)

  val_idx <- if (!is.null(val_cap) && nrow(validation) > val_cap) {
    sort(sample.int(nrow(validation), val_cap))
  } else {
    seq_len(nrow(validation))
  }
  val_sub <- validation[val_idx]

  t0 <- Sys.time()
  fit <- fit_fn(train_sub)
  fit_seconds <- as.numeric(difftime(Sys.time(), t0, units = "secs"))

  prob <- predict_fn(fit, val_sub)
  metrics <- compute_metrics(val_sub$isReturned, prob)
  metrics[, `:=`(model_id = id, family = family, train_n = nrow(train_sub), val_n = nrow(val_sub), fit_seconds = fit_seconds)]

  saveRDS(fit, file.path(MODEL_ZOO_DIR, paste0(id, ".rds")))
  cat("Done:", id, "-- AUC:", round(metrics$auc, 4), "| fit time:", round(fit_seconds, 1), "s\n")

  rm(train_sub, val_sub, fit)
  invisible(gc(verbose = FALSE))
  list(metrics = metrics, prob = prob, idx = val_idx)
}
```

```{r fit-fns-linear}
fit_glmnet_selected <- function(train_sub, alpha) {
  n <- nrow(train_sub)
  tune_idx <- sample.int(n, size = round(0.2 * n))
  fit_idx <- setdiff(seq_len(n), tune_idx)

  x_fit <- build_onehot(train_sub[fit_idx], LINEAR_PREDICTORS)
  y_fit <- train_sub$isReturned[fit_idx]
  path <- glmnet(x_fit, y_fit, family = "binomial", alpha = alpha, standardize = TRUE)

  x_tune <- build_onehot(train_sub[tune_idx], LINEAR_PREDICTORS)
  y_tune <- train_sub$isReturned[tune_idx]
  pred_path <- predict(path, newx = x_tune, type = "response")
  briers <- apply(pred_path, 2, function(p) brier_score(y_tune, p))
  best_lambda <- path$lambda[which.min(briers)]

  x_full <- build_onehot(train_sub, LINEAR_PREDICTORS)
  final_model <- glmnet(x_full, train_sub$isReturned, family = "binomial", alpha = alpha, lambda = best_lambda, standardize = TRUE)
  list(model = final_model, lambda = best_lambda, predictors = LINEAR_PREDICTORS, train_colnames = colnames(x_full))
}

}

  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  naivebayes::naive_bayes(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                           data = train_sub, usekernel = TRUE)
}
predict_naive_bayes <- function(fit, newdata) as.numeric(predict(fit, newdata = newdata, type = "prob")[, "Yes"])

# KNN is a lazy learner -- the "fit" is the frozen training subsample + k;
# kknn() both fits and predicts in one call. See the memory-profile note
# above for why its caps are much smaller than every other model's.
fit_knn <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  list(train_data = train_sub, k = 15)
}
predict_knn <- function(fit, newdata) {
  nd <- copy(newdata)
  nd[, y_factor := factor(NA, levels = c("No", "Yes"))]
  res <- kknn::kknn(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                     train = fit$train_data, test = nd, k = fit$k, kernel = "optimal", scale = TRUE)
  as.numeric(res$prob[, "Yes"])
}

fit_svm <- function(train_sub) {
  train_sub[, y_factor := factor(isReturned, levels = c(0, 1), labels = c("No", "Yes"))]
  e1071::svm(as.formula(paste("y_factor ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
             data = train_sub, kernel = "linear", probability = TRUE, scale = TRUE)
}
predict_svm <- function(fit, newdata) {
  pred <- predict(fit, newdata = newdata, probability = TRUE)
  as.numeric(attr(pred, "probabilities")[, "Yes"])
}

fit_nnet <- function(train_sub) {
  scale_params <- fit_numeric_scale(train_sub, feature_selector$linear_numeric_features)
  train_scaled <- apply_numeric_scale(train_sub, scale_params)
  model <- nnet::nnet(as.formula(paste("isReturned ~", paste(LINEAR_PREDICTORS, collapse = " + "))),
                       data = train_scaled, size = 8, decay = 0.01, maxit = 200, linout = FALSE, trace = FALSE)
  list(model = model, scale_params = scale_params)
}
predict_nnet <- function(fit, newdata) {
  nd_scaled <- apply_numeric_scale(newdata, fit$scale_params)
  as.numeric(predict(fit$model, newdata = nd_scaled, type = "raw"))
}
```

```{r run-all-models}
validation_results <- validation[, .(customer_id, variant_id, isReturned, value_exposure)]

model_zoo_results <- list()
model_zoo_failures <- list()

for (spec in model_zoo_spec) {
  result <- tryCatch(
    run_model(spec$id, spec$family, spec$fit_fn, spec$predict_fn, spec$train_cap, spec$val_cap),
    error = function(e) {
      cat("FAILED:", spec$id, "--", conditionMessage(e), "\n")
      model_zoo_failures[[length(model_zoo_failures) + 1]] <<- data.table(model_id = spec$id, error = conditionMessage(e))
      NULL
    }
  )
  if (!is.null(result)) {
    model_zoo_results[[length(model_zoo_results) + 1]] <- result$metrics
    validation_results[result$idx, (paste0("prob_", spec$id)) := result$prob]
  }
}

leaderboard <- rbindlist(model_zoo_results, fill = TRUE)

failures <- if (length(model_zoo_failures) > 0) rbindlist(model_zoo_failures) else data.table(model_id = character(0), error = character(0))

leaderboard
if (nrow(failures) > 0) { cat("\nFailed model(s):\n"); print(failures) }
```
